class IndexPattern(Model, frozen=True):
    """Substitute ascending bank ranks, then offset each selected pitch grid index.

    Intervals follow octave expansion and precede scale mapping and tuning.
    Omission leaves pitches unchanged. Repeats share a pattern entry; ordinary
    rests do not consume one. An outside-rank rest consumes its entry.
    """

    kind: Literal['index_pattern'] = 'index_pattern'
    indices: list[Annotated[int, Field(strict=True, ge=0)]] = Field(min_length=1)
    boundary: Literal['wrap', 'rest'] = 'wrap'
    intervals: (
        list[Annotated[int, Field(strict=True, ge=-(2**63), le=2**63 - 1)]] | None
    ) = None

    @model_validator(mode='after')
    def matching_intervals(self) -> Self:
        if self.intervals is not None and len(self.intervals) != len(self.indices):
            raise ValueError('intervals must have one value per index')
        return self

class Grammar(Model, frozen=True):
    kind: Literal['grammar'] = 'grammar'
    axiom: list[Annotated[str, Field(min_length=1)]] = Field(
        min_length=1, max_length=4096
    )
    rules: dict[
        Annotated[str, Field(min_length=1)], list[Annotated[str, Field(min_length=1)]]
    ] = Field(default_factory=dict)
    terminals: dict[
        Annotated[str, Field(min_length=1)],
        Annotated[int, Field(strict=True, ge=0, le=2**63 - 1)],
    ]
    rounds: int = Field(default=0, strict=True, ge=0, le=32)
    boundary: Literal['wrap', 'rest'] = 'wrap'

    def expand(self) -> IndexPattern:
        symbols = self.axiom
        for _ in range(self.rounds):
            expanded: list[str] = []
            for symbol in symbols:
                replacement = self.rules.get(symbol, [symbol])
                if len(expanded) + len(replacement) > 4096:
                    raise ValueError('grammar expansion exceeds 4096 symbols')
                expanded.extend(replacement)
            symbols = expanded
        if not symbols:
            raise ValueError('grammar expansion must not be empty')
        if any(s not in self.terminals for s in symbols):
            raise ValueError('every final grammar symbol requires a terminal rank')
        return IndexPattern(
            indices=[self.terminals[s] for s in symbols], boundary=self.boundary
        )

    @model_validator(mode='after')
    def valid_expansion(self) -> Self:
        self.expand()
        return self

class RankRange(Model, frozen=True):
    min: int = Field(strict=True, ge=-(2**63), le=2**63 - 1)
    max: int = Field(strict=True, ge=-(2**63), le=2**63 - 1)

    @model_validator(mode='after')

        if self.min > self.max:
            raise ValueError('range min must not exceed max')
        if self.max - self.min >= 2**64 - 1:

        return self

class Arpeggiator(Model):
    channel: int = Field(
        default=1, strict=True, ge=1, le=16, exclude_if=lambda c: c == 1
    )
    strum: Strum | None = None
    lanes: dict[Identifier, Arpeggiator] | None = Field(
        default=None, min_length=1, max_length=15
    )
    bank: Annotated[
        HeldBank | LatchedBank | HistoryBank | PhraseBank | RegionsBank,
        Field(discriminator='kind'),
    ] = HeldBank()
    selection: Annotated[
        Ascending
        | Descending
        | Played
        | Alternating
        | InsideOut
        | OutsideIn
        | IndexPattern
        | Grammar
        | Shuffle
        | Choice
        | Transition
        | IntervalEnvelope
        | Walk,
        Field(discriminator='kind'),
    ] = Ascending()
    rhythm: Annotated[
        Grid | Euclidean | SourceRhythm | Pattern, Field(discriminator='kind')
    ]
    gate: Rational = Field(default=Fraction(4, 5), ge=0)
    patterns: StepPatterns | None = None
    probability: Rational = Field(default=Fraction(1), ge=0, le=1)
    transposition: Transposition = Transposition()
    selection_offset: SelectionOffset = SelectionOffset()
    octaves: Octaves = Octaves()
    register_limits: PitchRegister | None = None
    scale: ScaleTransform | None = None
    tuning: Tuning | None = None
    bend_range: float | None = Field(default=None, gt=0)
    inversion: int = Field(default=0, strict=True, ge=-(2**63), le=2**63 - 1)
    voicing: (
        list[Annotated[int, Field(strict=True, ge=-(2**63), le=2**63 - 1)]] | None
    ) = Field(default=None, min_length=1)
    retrigger: Literal['on_empty', 'bank_edit'] = 'on_empty'
    expression: Expression = Expression()
    seed: int | None = Field(default=None, strict=True)

    @model_validator(mode='after')
    def seeded_probability(self) -> Self:
        if self.strum is not None:
            if not isinstance(self.bank, (HeldBank, LatchedBank)) or not isinstance(
                self.selection, (Ascending, Descending, Played)
            ):
                raise ValueError(
                    'strums require a held or latched bank and classic selection'
                )
        if self.lanes is not None:
            if not isinstance(self.bank, (HeldBank, LatchedBank)):
                raise ValueError('multiple lanes require held or latched banks')
            channels = [self.channel]
            for lane in self.lanes.values():
                if lane.lanes is not None or not isinstance(
                    lane.bank, (HeldBank, LatchedBank)
                ):
                    raise ValueError(
                        'additional lanes need held/latched banks without nested lanes'
                    )
                channels.append(lane.channel)
            if len(channels) != len(set(channels)):
                raise ValueError('each lane requires a distinct MIDI channel')
        if self.inversion and not isinstance(self.bank, (HeldBank, LatchedBank)):
            raise ValueError('inversion requires a held or latched bank')
        if self.voicing is not None and not isinstance(
            self.bank, (HeldBank, LatchedBank)
        ):
            raise ValueError('voicing requires a held or latched bank')
        if self.tuning is not None:
            if self.bend_range is None:
                raise ValueError(
                    'tuning requires an explicit MIDI bend_range in semitones'
                )
            source = self.tuning.source
            repeating = (
                isinstance(source, Computed)
                or (isinstance(source, RatioTable) and source.repeat_ratio is not None)
                or (isinstance(source, IntervalPattern) and source.repeat)
            )
            transforms = (
                self.inversion or any(self.octaves.offsets) or any(self.voicing or [])
            )
            folding = (
                self.register_limits.boundary
                if self.register_limits is not None
                else self.transposition.boundary
            ) == 'fold'
            if not repeating and (transforms or folding):
                raise ValueError(
                    'finite tunings cannot use period-based transforms or folding'
                )
        elif self.bend_range is not None:
            raise ValueError('bend_range requires tuning')
        random_walk = isinstance(self.selection, Walk) and len(self.selection.moves) > 1
        if (
            0 < self.probability < 1
            or random_walk
            or isinstance(
                self.selection, (Shuffle, Choice, Transition, IntervalEnvelope)
            )
        ) and self.seed is None:
            raise ValueError('random probability or selection requires a seed')
        return self

        repeating = (
            isinstance(source, Computed)
            or (isinstance(source, RatioTable) and source.repeat_ratio is not None)
            or (isinstance(source, IntervalPattern) and source.repeat)
        )
        transforms = (
            self.inversion or any(self.octaves.offsets) or any(self.voicing or [])
        )
        folding = (
            self.register_limits.boundary
            if self.register_limits is not None
            else self.transposition.boundary
        ) == 'fold'
        if not repeating and (transforms or folding):
            raise ValueError(
                'finite tunings cannot use period-based transforms or folding'
            )
    elif self.bend_range is not None:
        raise ValueError('bend_range requires tuning')
    random_walk = isinstance(self.selection, Walk) and len(self.selection.moves) > 1
    if (
        0 < self.probability < 1
        or random_walk
        or isinstance(
            self.selection, (Shuffle, Choice, Transition, IntervalEnvelope)
        )
    ) and self.seed is None:
        raise ValueError('random probability or selection requires a seed')
    return self
